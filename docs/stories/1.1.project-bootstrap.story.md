### Complete Prisma Schema (DM-1 to DM-9)
[Source: spec.md#section-3-4 + requirements.json domainModel]

```prisma
// prisma/schema.prisma

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// DM-8: Supplier
model Supplier {
  id          String    @id @default(cuid())
  name        String
  cnpj        String?
  contactName String?
  leadTimeDays Int      @default(7)
  products    Product[]
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt
}

// DM-1: Product (soft delete via isArchived)
model Product {
  id          String           @id @default(cuid())
  name        String
  brand       String
  description String?
  isArchived  Boolean          @default(false)
  supplier    Supplier?        @relation(fields: [supplierId], references: [id])
  supplierId  String?
  variants    ProductVariant[]
  createdAt   DateTime         @default(now())
  updatedAt   DateTime         @updatedAt
}

// DM-2: ProductVariant — all monetary fields are Decimal(10,2)
model ProductVariant {
  id           String               @id @default(cuid())
  sku          String               @unique
  frameColor   String
  lensColor    String
  uvProtection String
  isPolarized  Boolean              @default(false)
  costPrice    Decimal              @db.Decimal(10, 2)
  salePrice    Decimal              @db.Decimal(10, 2)
  images       String[]
  product      Product              @relation(fields: [productId], references: [id])
  productId    String
  inventory    Inventory?
  saleItems    SaleItem[]
  createdAt    DateTime             @default(now())
  updatedAt    DateTime             @updatedAt
}

// DM-3: Inventory [1:1 with ProductVariant]
model Inventory {
  id             String                 @id @default(cuid())
  quantity       Int                    @default(0)
  minStockAlert  Int                    @default(3)
  variant        ProductVariant         @relation(fields: [variantId], references: [id])
  variantId      String                 @unique
  transactions   InventoryTransaction[]
  updatedAt      DateTime               @updatedAt
}

// DM-4: InventoryTransaction — append-only audit trail
model InventoryTransaction {
  id            String    @id @default(cuid())
  type          TxType
  quantityDelta Int
  note          String?
  inventory     Inventory @relation(fields: [inventoryId], references: [id])
  inventoryId   String
  createdAt     DateTime  @default(now())
}

enum TxType {
  SALE
  PURCHASE
  RETURN
  EXCHANGE
  ADJUSTMENT
}

// DM-7: Customer (optional link in Sale)
model Customer {
  id        String   @id @default(cuid())
  name      String
  phone     String?
  email     String?
  tags      String[]
  sales     Sale[]
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}

// DM-5: Sale — totalAmount and payment fields are Decimal
model Sale {
  id               String        @id @default(cuid())
  totalAmount      Decimal       @db.Decimal(10, 2)
  paymentBreakdown Json
  status           SaleStatus    @default(COMPLETED)
  refNfe           String?
  customer         Customer?     @relation(fields: [customerId], references: [id])
  customerId       String?
  items            SaleItem[]
  cashFlowEntry    CashFlowEntry?
  createdAt        DateTime      @default(now())
  updatedAt        DateTime      @updatedAt
}

enum SaleStatus {
  COMPLETED
  CANCELLED
  RETURNED
}

// DM-6: SaleItem — unitPrice/unitCost are snapshots at time of sale
model SaleItem {
  id        String         @id @default(cuid())
  quantity  Int
  unitPrice Decimal        @db.Decimal(10, 2)
  unitCost  Decimal        @db.Decimal(10, 2)
  sale      Sale           @relation(fields: [saleId], references: [id])
  saleId    String
  variant   ProductVariant @relation(fields: [variantId], references: [id])
  variantId String
}

// DM-9: CashFlowEntry [1:1 optional with Sale for automatic income entries]
model CashFlowEntry {
  id     String          @id @default(cuid())
  type   CashFlowType
  amount Decimal         @db.Decimal(10, 2)
  note   String?
  date   DateTime        @default(now())
  sale   Sale?           @relation(fields: [saleId], references: [id])
  saleId String?         @unique
}

enum CashFlowType {
  INCOME
  EXPENSE
}
```

**Critical schema rules:**
- `isArchived` on Product — soft delete only, never physical DELETE if has sales (enforced in service layer)
- `InventoryTransaction` — append-only, no UPDATE/DELETE ever
- `SaleItem.unitPrice` + `SaleItem.unitCost` — snapshots captured at time of sale, immutable
- `paymentBreakdown: Json` — extensible without migration (stores `{pix, cardCredit, cardDebit, cash}`)
- `refNfe: String?` — reserved for future NF-e integration, nullable now (ASM-4)

### Decimal.js — Critical Pattern
[Source: spec.md#section-3-1 + architecture-decision.json]

```typescript
// src/lib/decimal.ts
import Decimal from 'decimal.js'

export function calculateMargin(salePrice: Decimal, costPrice: Decimal): Decimal {
  if (salePrice.isZero()) return new Decimal(0)
  return salePrice.minus(costPrice).dividedBy(salePrice).times(100)
}

export function formatCurrency(value: Decimal): string {
  return new Intl.NumberFormat('pt-BR', {
    style: 'currency',
    currency: 'BRL',
  }).format(value.toNumber())
}
```

> **Why Decimal.js:** `0.1 + 0.2 = 0.30000000000000004` in JavaScript Float. For a store with 100 variants at R$150 each, Float rounding accumulates to real money loss over time. [Source: spec.md#section-3-1 rationale financeiro crítico]
