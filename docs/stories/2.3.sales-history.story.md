### processReturn — Atomic Flow
[Source: spec.md#section-3.3 — critical operation pattern; FR-9 AC-9.1, AC-9.2]

```typescript
// src/features/sales/services/sales.service.ts
static async processReturn(saleId: string): Promise<SaleWithItems> {
  // Fetch full sale (reuses getSaleById which throws if not found)
  const sale = await SalesService.getSaleById(saleId)

  // Idempotency guard — BEFORE $transaction (no wasted DB round-trip)
  if (sale.status !== 'COMPLETED') {
    throw new Error('Venda já foi devolvida')
  }

  return prisma.$transaction(async (tx) => {
    // Step a: mark sale RETURNED
    await tx.sale.update({ where: { id: saleId }, data: { status: 'RETURNED' } })

    for (const item of sale.items) {
      // Need inventory ID — fetch via variantId
      const inv = await tx.inventory.findUniqueOrThrow({ where: { variantId: item.variantId } })

      // Step b: restore stock (atomic increment — no read-then-write race)
      await tx.inventory.update({
        where: { id: inv.id },
        data: { quantity: { increment: item.quantity } },
      })

      // Step c: audit trail
      await tx.inventoryTransaction.create({
        data: { inventoryId: inv.id, type: 'RETURN', quantityDelta: item.quantity },
      })
    }

    // Step d: financial reversal (outflow = refund)
    await tx.cashFlowEntry.create({
      data: { type: 'EXPENSE', amount: sale.totalAmount.toString(), saleId },
    })

    return tx.sale.findUniqueOrThrow({
      where: { id: saleId },
      include: {
        items: { include: { variant: { include: { product: { select: { name: true } } } } } },
        customer: true,
        cashFlowEntry: true,
      },
    }) as Promise<SaleWithItems>
  })
}
```

**Critical:** use `{ increment: item.quantity }` (Prisma atomic update), NOT a fetch+update pattern — avoids race condition if two concurrent returns are attempted.

### Server Actions Pattern
[Source: src/features/sales/actions.ts — established pattern from Story 2.1]

```typescript
// Add to src/features/sales/actions.ts
'use server'

export async function returnSaleAction(
  saleId: string,
): Promise<{ success: true; saleId: string } | { success: false; error: string }> {
  try {
    const sale = await SalesService.processReturn(saleId)
    revalidatePath('/vendas')
    revalidatePath('/estoque')
    return { success: true, saleId: sale.id }
  } catch (err) {
    return { success: false, error: err instanceof Error ? err.message : 'Erro ao processar devolução' }
  }
}

export async function getSaleByIdAction(
  saleId: string,
): Promise<{ success: true; sale: SaleWithItems } | { success: false; error: string }> {
  try {
    const sale = await SalesService.getSaleById(saleId)
    return { success: true, sale }
  } catch (err) {
    return { success: false, error: err instanceof Error ? err.message : 'Erro ao carregar venda' }
  }
}
```

### getSaleByIdAction — Lazy Fetch Pattern

The sale detail is loaded on-demand when the user clicks a row. The component:
1. Stores `selectedSale: SaleWithItems | null` in local state
2. On row click: sets `isLoadingDetail = true`, calls `getSaleByIdAction(saleId)`, sets `selectedSale`, opens Sheet
3. Sheet shows skeleton/spinner while `isLoadingDetail`

This avoids pre-fetching all sale details (N+1) at page load.
